#+begin_src python
import nltk
from sklearn.feature_extraction.text import CountVectorizer
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer
import numpy as np

nltk.download('punkt')
nltk.download('stopwords')
nltk.download('wordnet')
#+end_src

** Define Keywords and Criteria
- Define a dictionary of keywords and their corresponding weights based on their importance for the job role.
- Define a sample input that is extracted from resume.
#+begin_src python
keyword_weights = {
    'python': 2,
    'machine learning': 3,
    'data analysis': 2,
    'deep learning': 3
}
keywords = list(keyword_weights.keys())

resumes = [
    "Experienced in Python and machine learning. Worked on data analysis projects.",
    "Skilled in deep learning and NLP. Proficient in Python.",
    "Expert in data analysis and machine learning. Familiar with NLP techniques."
]
#+end_src

** Text Preprocessing
- Define a preprocessing function to:
  - Tokenize the text into words.
  - Convert text to lowercase.
  - Remove stopwords and non-alphanumeric tokens.
  - Lemmatize tokens to their base form.
- Apply the preprocessing function to all resumes.
  
#+begin_src python
def preprocess(text):
    lemmatizer = WordNetLemmatizer()
    stop_words = set(stopwords.words('english'))
    tokens = nltk.word_tokenize(text.lower())
    tokens = [lemmatizer.lemmatize(token) for token in tokens if token.isalnum() and token not in stop_words]
    return ' '.join(tokens)

preprocessed_resumes = [preprocess(resume) for resume in resumes]
#+end_src

** Feature Extraction
- Use CountVectorizer to convert the preprocessed resumes into a binary matrix of keyword presence.
- Set `ngram_range=(1, 2)` to capture both single words and phrases (e.g., "machine learning").
- Fit and transform the resumes into a feature matrix.
  
#+begin_src python
vectorizer = CountVectorizer(vocabulary=keywords, ngram_range=(1, 2), binary=True)
count_matrix = vectorizer.fit_transform(preprocessed_resumes).toarray()
#+end_src

** Scoring and Ranking
- Calculate scores for each resume by multiplying the binary keyword presence matrix with the keyword weights.
- Rank the resumes based on their scores in descending order.
- Print the ranked resumes along with their scores and the total possible score.

#+begin_src python
weights = np.array([keyword_weights[key] for key in keywords])
scores = np.dot(count_matrix, weights)
ranked_resumes = sorted(zip(resumes, scores), key=lambda x: x[1], reverse=True)

total_possible_score = sum(keyword_weights.values())

for idx, (resume, score) in enumerate(ranked_resumes, start=1):
    print(f"Rank {idx} - Score: {score}/{total_possible_score}\nResume: {resume}\n")
#+end_src
